Instructor / Host Configuration

Connect Google Sheets & Automated Gmail Receipts

Follow these 4 steps to log every student sign-up into a Google Sheet and automatically send confirmation emails to both you and the student.

  1. Create a new Google Sheet and open Extensions → Apps Script.
  2. Paste the script below (update HOST_EMAIL and PAYMENT_INSTRUCTIONS) and click Save.
  3. Click Deploy → New deployment → Web app. Set Execute as: Me and Who has access: Anyone.
  4. Copy the generated Web App URL and paste it into GOOGLE_SCRIPT_URL at the top of the <script> in index.html.
Code.gs (Google Apps Script) 
const HOST_EMAIL = "your-email@gmail.com"; // Replace with your email
const SHEET_NAME = "Registrations";

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
      sheet.appendRow([
        "Timestamp", "Reg ID", "Full Name", "Email",
        "Phone", "Affiliation", "Track", "Project Interest", "Target Language"
      ]);
    }

    const regId = data.regId || ("ASR-" + Math.floor(100000 + Math.random() * 900000));
    sheet.appendRow([
      data.submittedAt || new Date().toISOString(),
      regId,
      data.fullName,
      data.email,
      data.phone,
      data.affiliation,
      data.track,
      data.projectInterest,
      data.targetLanguage || ""
    ]);

    const isCertified = (data.track || "").indexOf("999") !== -1;
    const feeNote = isCertified
      ? "You selected the Certified Internship Track (Rs 999). We will share the official payment link with you shortly via Email/WhatsApp. Your certificate will be issued upon assignment completion and fee verification."
      : "You selected the Free Learner Track (Rs 0). You have full access to all YouTube lectures and Google Colab notebooks on the course portal.";

    // 1. Send confirmation email to the Student
    MailApp.sendEmail({
      to: data.email,
      subject: "Registration Confirmed: Low-Resource ASR School & Internship (" + regId + ")",
      body: "Dear " + data.fullName + ",\n\n" +
            "Thank you for registering for the Low-Resource ASR School & Internship!\n\n" +
            "Registration ID: " + regId + "\n" +
            "Selected Track: " + data.track + "\n" +
            "Affiliation: " + data.affiliation + "\n" +
            "Project Interest: " + data.projectInterest + "\n\n" +
            feeNote + "\n\n" +
            "Best regards,\nLow-Resource ASR School Team"
    });

    // 2. Send notification email to the Host
    MailApp.sendEmail({
      to: HOST_EMAIL,
      subject: "[New ASR Sign-Up] " + data.fullName + " - " + data.track,
      body: "New participant registered:\n\n" +
            "Reg ID: " + regId + "\n" +
            "Name: " + data.fullName + "\n" +
            "Email: " + data.email + "\n" +
            "Phone: " + data.phone + "\n" +
            "Affiliation: " + data.affiliation + "\n" +
            "Track: " + data.track + "\n" +
            "Project: " + data.projectInterest + "\n" +
            "Language: " + (data.targetLanguage || "N/A")
    });

    return ContentService.createTextOutput(JSON.stringify({ status: "success", regId: regId }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
